class MySecurity : public BLESecurityCallbacks
{
    uint32_t onPassKeyRequest()
    {
        Serial.println("Solicitacao de PIN");
        return 133700;
    }

    void onPassKeyNotify(uint32_t pass_key)
    {
        Serial.print("PIN recebido: ");
        Serial.println(pass_key);
    }

    bool onConfirmPIN(uint32_t pass_key)
    {
        Serial.print("Confirmando PIN: ");
        Serial.println(pass_key);
        return true;
    }

    bool onSecurityRequest()
    {
        Serial.println("Solicitacao de seguranca");
        return true;
    }

    void onAuthenticationComplete(esp_ble_auth_cmpl_t auth_cmpl)
    {
        if (auth_cmpl.success)
        {
            Serial.println("Autenticacao BLE concluida.");
            paired = true;
        }
        else
        {
            Serial.println("Falha na autenticacao BLE.");
            paired = false;
        }
    }
};

// ============================================================
// CALLBACKS DE CONEXÃO BLE
// (detecta quando a câmera cai sozinha, ex: dormiu ou saiu de alcance)
// ============================================================

class MyClientCallback : public BLEClientCallbacks
{
    void onConnect(BLEClient* client) {}

    void onDisconnect(BLEClient* client)
    {
        connected = false;
        paired = false;

        pCommandCharacteristic = nullptr;
        pNotifyCharacteristic = nullptr;

        Serial.println("Camera desconectou (evento BLE).");
    }
};

// ============================================================
// NOTIFICAÇÕES DA SONY
// ============================================================

void notifyCallback(
    BLERemoteCharacteristic* characteristic,
    uint8_t* data,
    size_t length,
    bool isNotify)
{
    Serial.println();
    Serial.println("----------------------------------------");
    Serial.println("RESPOSTA DA ZV-E10 - FF02");
    Serial.print("HEX: ");

    for (size_t i = 0; i < length; i++)
    {
        if (data[i] < 0x10)
            Serial.print("0");

        Serial.print(data[i], HEX);
        Serial.print(" ");
    }

    Serial.println();
    Serial.println("----------------------------------------");
}

// ============================================================

bool connectToCamera()
{
    if (connected)
    {
        Serial.println("Camera ja esta conectada.");
        return true;
    }

    Serial.println();
    Serial.println("Procurando ZV-E10...");

    BLEScan* pScan = BLEDevice::getScan();

    pScan->setActiveScan(true);

    BLEScanResults results = pScan->start(10, false);

    BLEAdvertisedDevice* cameraDevice = nullptr;

    for (int i = 0; i < results.getCount(); i++)
    {
        BLEAdvertisedDevice device = results.getDevice(i);

        String name = device.getName().c_str();

        Serial.print("Dispositivo: ");
        Serial.println(name);

        if (name.indexOf(CAMERA_NAME) >= 0)
        {
            Serial.println("ZV-E10 encontrada!");

            cameraDevice = new BLEAdvertisedDevice(device);
            break;
        }
    }

    // Libera a memoria dos resultados do scan (evita acumulo a cada reconexao)
    pScan->clearResults();

    if (cameraDevice == nullptr)
    {
        Serial.println("Camera nao encontrada.");
        return false;
    }

    Serial.println("Conectando...");

    pClient = BLEDevice::createClient();
    pClient->setClientCallbacks(new MyClientCallback());

    bool ok = pClient->connect(cameraDevice);

    // connect() ja copia o endereco internamente, o ponteiro pode ser liberado
    delete cameraDevice;

    if (!ok)
    {
        Serial.println("Falha ao conectar.");
        return false;
    }

    Serial.println("Conectado à camera.");

    BLERemoteService* pService =
        pClient->getService(BLEUUID(SONY_SERVICE_UUID));

    if (pService == nullptr)
    {
        Serial.println("Servico Sony nao encontrado.");
        pClient->disconnect();
        return false;
    }

    pCommandCharacteristic =
        pService->getCharacteristic(BLEUUID(SONY_COMMAND_UUID));

    pNotifyCharacteristic =
        pService->getCharacteristic(BLEUUID(SONY_NOTIFY_UUID));

    if (pCommandCharacteristic == nullptr)
    {
        Serial.println("Caracteristica FF01 nao encontrada.");
        pClient->disconnect();
        return false;
    }

    if (pNotifyCharacteristic != nullptr && pNotifyCharacteristic->canNotify())
    {
        pNotifyCharacteristic->registerForNotify(notifyCallback);
        Serial.println("FF02 configurado.");
    }

    connected = true;

    Serial.println("Camera pronta.");

    return true;
}

void disconnectFromCamera()
{
    if (pClient != nullptr && connected)
    {
        pClient->disconnect();
    }

    connected = false;
    paired = false;

    pCommandCharacteristic = nullptr;
    pNotifyCharacteristic = nullptr;

    Serial.println("Camera desconectada.");
}

// Envia um comando e confirma que a camera ainda esta conectada antes disso.
// Retorna false se a sequencia deve ser abortada.
bool sendShutterCommand(const uint8_t* cmd, size_t len, const char* label)
{
    if (!connected || pCommandCharacteristic == nullptr)
    {
        Serial.print("Sequencia abortada antes de enviar: ");
        Serial.println(label);
        return false;
    }

    Serial.print("Enviando ");
    Serial.println(label);

    pCommandCharacteristic->writeValue((uint8_t*)cmd, len, true);

    return true;
}

void takePhoto()
{
    if (shooting)
    {
        Serial.println("Disparo ignorado: sequencia em andamento.");
        return;
    }

    if (!connected || pCommandCharacteristic == nullptr)
    {
        Serial.println("Camera nao conectada.");
        return;
    }

    if (millis() - lastShotTime < minShotDelay)
    {
        Serial.println("Disparo ignorado: intervalo minimo ainda nao terminou.");
        return;
    }

    shooting = true;

    Serial.println();
    Serial.println("========================================");
    Serial.println("DISPARANDO FOTO");
    Serial.println("========================================");

    bool ok = true;

    uint8_t halfDown[] = {0x01, 0x07};
    ok = sendShutterCommand(halfDown, sizeof(halfDown), "Shutter Half Down: 01 07");
    if (ok) delay(halfDownDelay);

    uint8_t fullyDown[] = {0x01, 0x09};
    if (ok) ok = sendShutterCommand(fullyDown, sizeof(fullyDown), "Shutter Fully Down: 01 09");
    if (ok) delay(fullDownDelay);

    uint8_t fullyUp[] = {0x01, 0x08};
    if (ok) ok = sendShutterCommand(fullyUp, sizeof(fullyUp), "Shutter Fully Up: 01 08");
    if (ok) delay(fullUpDelay);

    uint8_t halfUp[] = {0x01, 0x06};
    if (ok) ok = sendShutterCommand(halfUp, sizeof(halfUp), "Shutter Half Up: 01 06");

    if (ok)
    {
        photoCount++;
        lastShotTime = millis();

        Serial.println("Foto concluida.");
        Serial.print("Total de fotos: ");
        Serial.println(photoCount);
    }
    else
    {
        Serial.println("Sequencia de disparo interrompida (camera desconectou no meio do processo).");
    }

    shooting = false;

    Serial.println("========================================");
}

// ============================================================
// BOTÃO
// ============================================================

void configureButtonPin()
{
    pinMode(buttonPin, INPUT_PULLUP);

    lastButtonReading = digitalRead(buttonPin);
    stableButtonState = lastButtonReading;

    Serial.print("Botao configurado no GPIO ");
    Serial.println(buttonPin);
}

void checkButton()
{
    bool reading = digitalRead(buttonPin);

    if (reading != lastButtonReading)
    {
        lastDebounceTime = millis();
    }

    if ((millis() - lastDebounceTime) > debounceDelay)
    {
        if (reading != stableButtonState)
        {
            stableButtonState = reading;

            if (stableButtonState == LOW)
            {
                Serial.println("Botao pressionado.");
                takePhoto();
            }
        }
    }

    lastButtonReading = reading;
}

void handleRoot()
{
    server.send(200, "text/html; charset=utf-8", htmlPage());

void handleSave()
{
    if (server.hasArg("buttonPin"))
        buttonPin = server.arg("buttonPin").toInt();

    if (server.hasArg("halfDown"))
        halfDownDelay = server.arg("halfDown").toInt();

    if (server.hasArg("fullDown"))
        fullDownDelay = server.arg("fullDown").toInt();

    if (server.hasArg("fullUp"))
        fullUpDelay = server.arg("fullUp").toInt();

    if (server.hasArg("debounce"))
        debounceDelay = server.arg("debounce").toInt();

    if (server.hasArg("minShot"))
        minShotDelay = server.arg("minShot").toInt();

    if (buttonPin < 0 || buttonPin > 39)
        buttonPin = DEFAULT_BUTTON_PIN;

    if (halfDownDelay > 10000)
        halfDownDelay = DEFAULT_HALF_DOWN_DELAY;

    if (fullDownDelay > 10000)
        fullDownDelay = DEFAULT_FULL_DOWN_DELAY;

    if (fullUpDelay > 10000)
        fullUpDelay = DEFAULT_FULL_UP_DELAY;

    if (debounceDelay < 10 || debounceDelay > 2000)
        debounceDelay = DEFAULT_DEBOUNCE_DELAY;

    if (minShotDelay > 10000)
        minShotDelay = DEFAULT_MIN_SHOT_DELAY;

    saveSettings();
    configureButtonPin();

    server.sendHeader("Location", "/");
    server.send(303);
}

    if (shooting)
    {
        server.send(409, "text/plain; charset=utf-8", "Disparo ignorado: outra sequência está em andamento.");
        return;
    }

    if (!connected)
    {
        server.send(503, "text/plain; charset=utf-8", "Câmera não conectada.");
        return;
    }

    takePhoto();

    server.send(200, "text/plain; charset=utf-8", "Disparo concluído.");
}

void handleConnect()
{
    bool result = connectToCamera();

    if (result)
    {
        server.send(200, "text/plain; charset=utf-8", "Câmera conectada.");
    }
    else
    {
        server.send(500, "text/plain; charset=utf-8", "Não foi possível conectar à câmera.");
    }
}

void handleDisconnect()
{
    disconnectFromCamera();
    server.send(200, "text/plain; charset=utf-8", "Câmera desconectada.");
}

void handleStatus()
{
    String json = "{";
    json += "\"connected\":";
    json += connected ? "true" : "false";
    json += ",\"shooting\":";
    json += shooting ? "true" : "false";
    json += ",\"photos\":";
    json += String(photoCount);
    json += "}";

    server.send(200, "application/json; charset=utf-8", json);
}

// Qualquer URL desconhecida redireciona para "/".
// É o truque padrão que faz Android/iOS/Windows abrirem
// a página de login automaticamente ao entrar no Wi-Fi do ESP32.
void handleNotFound()
{
    server.sendHeader("Location", "/", true);
    server.send(302, "text/plain", "");
}

// ============================================================
// CONFIGURAR WI-FI, DNS E SERVIDOR
// ============================================================

void setupWebServer()
{
    WiFi.mode(WIFI_AP);
    WiFi.softAP(AP_SSID, AP_PASSWORD);

    // Reduz interferencia do Wi-Fi no radio compartilhado com o BLE
    WiFi.setSleep(false);

    IPAddress apIP = WiFi.softAPIP();

    dnsServer.start(DNS_PORT, "*", apIP);

    Serial.println();
    Serial.println("========================================");
    Serial.println("WI-FI CONFIGURADO");
    Serial.print("Rede: ");
    Serial.println(AP_SSID);
    Serial.print("IP: ");
    Serial.println(apIP);
    Serial.println("========================================");

    server.on("/", HTTP_GET, handleRoot);
    server.on("/save", HTTP_POST, handleSave);
    server.on("/photo", HTTP_POST, handlePhoto);
    server.on("/connect", HTTP_POST, handleConnect);
    server.on("/disconnect", HTTP_POST, handleDisconnect);
    server.on("/status", HTTP_GET, handleStatus);
    server.onNotFound(handleNotFound);

    server.begin();

    Serial.println("Servidor web iniciado.");
}

// ============================================================
// SETUP
// ============================================================

void setup()
{
    Serial.begin(115200);

    // Endereco Bluetooth novo: a camera guardou uma chave antiga para o
    // endereco original e recusa a autenticacao. Com outro endereco ela
    // ve um controle remoto novo e pareia do zero.
    // (precisa vir antes de iniciar Bluetooth e Wi-Fi)
    uint8_t mac[6];
    esp_efuse_mac_get_default(mac);
    mac[0] |= 0x02;     // endereco "administrado localmente", diferente do de fabrica
    esp_base_mac_addr_set(mac);

    delay(1000);

    Serial.println();
    Serial.println("========================================");
    Serial.println("SONY ZV-E10 - ESP32 REMOTE V7");
    Serial.println("========================================");

    loadSettings();
    configureButtonPin();

    BLEDevice::init("");
    BLEDevice::setEncryptionLevel(ESP_BLE_SEC_ENCRYPT);
    BLEDevice::setSecurityCallbacks(new MySecurity());

    setupWebServer();

    connectToCamera();

    Serial.println();
    Serial.println("Comandos seriais:");
    Serial.println("F = disparar foto");
    Serial.println("C = conectar");
    Serial.println("D = desconectar");
    Serial.println("========================================");
}

// ============================================================
// LOOP
// ============================================================

void loop()
{
    dnsServer.processNextRequest();
    server.handleClient();

    checkButton();

    if (Serial.available())
    {
        char command = Serial.read();

        if (command == 'F' || command == 'f')
        {
            takePhoto();
        }
        else if (command == 'C' || command == 'c')
        {
            connectToCamera();
        }
        else if (command == 'D' || command == 'd')
        {
            disconnectFromCamera();
        }
    }

    delay(5);
}
